cmake_minimum_required(VERSION 3.16)
project(CacheLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cache_simulator cache_simulator.cpp)
add_executable(target_workload target_workload.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cache_simulator PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(target_workload PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME simulator_regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/test_driver.py $<TARGET_FILE:cache_simulator> --workload $<TARGET_FILE:target_workload>)
endif()
